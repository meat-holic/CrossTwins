function(settings_body out file)
  file(READ "${REPO_ROOT}/${file}" body)
  string(REGEX REPLACE "#[ \t]*include[^\n]*\n" "" body "${body}")
  set(${out} "${body}" PARENT_SCOPE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REPO_ROOT}/${file}")
endfunction()
function(settings_slice out file begin end)
  file(READ "${REPO_ROOT}/${file}" body)
  string(FIND "${body}" "${begin}" first)
  string(FIND "${body}" "${end}" last)
  if(first LESS 0 OR last LESS_EQUAL first)
    message(FATAL_ERROR "Settings slice not found: ${file}")
  endif()
  math(EXPR length "${last}-${first}")
  string(SUBSTRING "${body}" ${first} ${length} body)
  set(${out} "${body}" PARENT_SCOPE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REPO_ROOT}/${file}")
endfunction()
